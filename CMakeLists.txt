cmake_minimum_required(VERSION 3.14)
project(meo_server CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(Threads REQUIRED)
add_executable(meo_server server/main.cpp)
target_include_directories(meo_server PRIVATE third_party)
target_link_libraries(meo_server PRIVATE Threads::Threads)
if(WIN32)
  target_link_libraries(meo_server PRIVATE ws2_32)
endif()
if(MSVC)
  target_compile_options(meo_server PRIVATE /W4 /utf-8)
else()
  target_compile_options(meo_server PRIVATE -Wall -Wextra)
endif()
